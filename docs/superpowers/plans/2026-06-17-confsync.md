# confsync Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Un singolo script Bash POSIX (`confsync`) che backuppa e ripristina in modo seamless e cifrato tutte le customizzazioni dell'OS su un bucket GCS, cross-distro e a costo di storage minimo.

**Architecture:** Script monolitico con sotto-comandi (`backup`/`restore`/`list`/`diff`). Pipeline backup: raccolta da manifest → `tar | zstd` → cifratura `openssl aes-256-cbc` client-side → dedup SHA256 → upload `gsutil`. Restore inverso con verifica integrità. Tre artefatti separati (dotfiles, secrets, repos) per isolare i segreti. Bucket Nearline single-region con versioning + lifecycle.

**Tech Stack:** Bash (POSIX-ish, bash 4+), `tar`, `zstd` (fallback `gzip`), `openssl` (AES-256-CBC + PBKDF2), Google Cloud SDK (`gcloud`/`gsutil`), `shellcheck` + `bats` per il test.

## Global Constraints

- **Security first**: nessun dato in chiaro lascia la macchina; cifratura `openssl enc -aes-256-cbc -pbkdf2 -iter 600000 -salt -pass stdin` SEMPRE prima dell'upload. Temp file solo sotto `mktemp -d` con `trap` di cleanup; mai temp persistenti in chiaro.
- **Cifratura**: `openssl` (NON `age` — age richiede TTY e non è scriptabile). Artefatti cifrati con estensione `.tar.enc`. Passphrase via env `CONFSYNC_PASSPHRASE` o prompt `read -rs`, mai salvata.
- **Cross-distro**: rilevamento via `/etc/os-release`; nessun assunto Manjaro-only.
- **Singolo file** `confsync` come deliverable principale; `set -euo pipefail` in testa.
- Bucket: `gs://il-tuo-bucket`, progetto `il-tuo-progetto-gcp`, account `tua.email@example.com`, location `europe-west1`, class **Nearline**, uniform BLA + public-access-prevention enforced, versioning ON.
- Layout oggetti: `gs://il-tuo-bucket/<hostname>/<timestamp>/{dotfiles.tar.enc,secrets.tar.enc,repo-localfiles.tar.enc,repos.manifest.json,packages.txt,MANIFEST.txt}` + puntatore `<hostname>/latest`.
- Passphrase unica per dotfiles e secrets, mai salvata; chiesta interattivamente o via `CONFSYNC_PASSPHRASE` (sconsigliato).
- Repo scan default: `~/code`. Dedup: niente re-upload se SHA256 dell'archivio compresso == ultimo backup.
- Manovalanza: scrittura delegata a **opencode**, orchestrata/rivista/testata da Claude Code.

`confsync` (scheletro):
```bash
#!/usr/bin/env bash
set -euo pipefail

BUCKET="${CONFSYNC_BUCKET:-gs://il-tuo-bucket}"
HOSTNAME_TAG="$(hostname)"
SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"

log()  { printf '>> %s\n' "$*" >&2; }
die()  { printf 'ERRORE: %s\n' "$*" >&2; exit 1; }

# Source manifest (fallback embedded minimale se assente)
if [ -f "$SCRIPT_DIR/lib/manifest.sh" ]; then
  # shellcheck source=lib/manifest.sh
  . "$SCRIPT_DIR/lib/manifest.sh"
else
  INCLUDE_DOTFILES=(.zshrc .bashrc .profile .config .claude CLAUDE.md .gitconfig)
  EXCLUDE_PATTERNS=('*/Cache/*' '*.log' '*/.git/*' '*/node_modules/*')
  SECRET_PATHS=(.ssh .gnupg .netrc)
  SECRET_PATTERNS=('.env' '*.key' '*.pem' '*secret*' '*token*')
  REPO_SCAN_DIRS=("$HOME/code")
fi

detect_distro() {
  local f="${CONFSYNC_OS_RELEASE:-/etc/os-release}"
  local id="" idlike=""
  if [ -r "$f" ]; then
    id="$(. "$f" 2>/dev/null; printf '%s' "${ID:-}")"
    idlike="$(. "$f" 2>/dev/null; printf '%s' "${ID_LIKE:-}")"
  fi
  case "$id $idlike" in
    *arch*)   echo arch ;;
    *debian*|*ubuntu*) echo debian ;;
    *fedora*|*rhel*)   echo fedora ;;
    *) echo unknown ;;
  esac
}

pkg_list_cmd() {
  case "$(detect_distro)" in
    arch)   echo "pacman -Qqe" ;;
    debian) echo "apt-mark showmanual" ;;
    fedora) echo "dnf repoquery --userinstalled --queryformat '%{name}'" ;;
    *)      echo "" ;;
  esac
}

require_deps() {
  local missing=()
  for d in tar openssl gsutil; do
    if [ "${CONFSYNC_FAKE_MISSING:-}" = "$d" ] || ! command -v "$d" >/dev/null 2>&1; then
      missing+=("$d")
    fi
  done
  [ ${#missing[@]} -eq 0 ] || die "dipendenze mancanti: ${missing[*]} (installa: openssl, zstd, google-cloud-sdk)"
}

main() {
  case "${1:-}" in
    --source-only) return 0 ;;
    backup)  shift; cmd_backup "$@" ;;
    restore) shift; cmd_restore "$@" ;;
    list)    shift; cmd_list "$@" ;;
    diff)    shift; cmd_diff "$@" ;;
    ""|-h|--help) usage ;;
    *) die "comando sconosciuto: $1 (usa: backup|restore|list|diff)" ;;
  esac
}

usage() { cat >&2 <<EOF
confsync — backup/restore cifrato delle config OS
  confsync backup
  confsync restore [--secrets] [--repos] [--packages] [--dry-run]
  confsync list
  confsync diff
EOF
}

main "$@"
```

- [ ] **Step 3: Implementare le funzioni in `confsync`** (sopra `main`)

```bash
compressor()   { if command -v zstd >/dev/null 2>&1; then echo "zstd -19 -T0"; else echo "gzip -9"; fi; }
decompressor() { if command -v zstd >/dev/null 2>&1; then echo "zstd -d"; else echo "gzip -d"; fi; }

get_passphrase() {
  if [ -n "${CONFSYNC_PASSPHRASE:-}" ]; then printf '%s' "$CONFSYNC_PASSPHRASE"; return 0; fi
  local p; read -rsp "Passphrase confsync: " p >&2; printf '\n' >&2; printf '%s' "$p"
}

sha256_of() { sha256sum "$1" | awk '{print $1}'; }

# Cifra/decifra leggendo la passphrase da fd 3 (-pass fd:3): mai in argv/env di openssl,
# mai un file temporaneo in chiaro con la passphrase.
_openssl_enc() { openssl enc -aes-256-cbc -pbkdf2 -iter 600000 -salt -pass fd:3 -out "$1" 3<<<"$2"; }
_openssl_dec() { openssl enc -d -aes-256-cbc -pbkdf2 -iter 600000 -pass fd:3 -in "$1" 3<<<"$2"; }

pack_encrypt() {
  local src="$1" out="$2"; shift 2
  local pass; pass="$(get_passphrase)"
  local existing=(); local p
  for p in "$@"; do [ -e "$src/$p" ] && existing+=("$p"); done
  [ ${#existing[@]} -gt 0 ] || { log "nessun path da archiviare per $out"; return 0; }
  # tar -> compressione -> openssl (passphrase da fd:3). out scritto da openssl.
  tar -C "$src" -cf - "${existing[@]}" \
    | $(compressor) \
    | openssl enc -aes-256-cbc -pbkdf2 -iter 600000 -salt -pass fd:3 -out "$out" 3<<<"$pass"
}

decrypt_unpack() {
  local in="$1" dest="$2"
  local pass; pass="$(get_passphrase)"
  mkdir -p "$dest"
  openssl enc -d -aes-256-cbc -pbkdf2 -iter 600000 -pass fd:3 -in "$in" 3<<<"$pass" \
    | $(decompressor) \
    | tar -C "$dest" -xf -
}
```

> Nota per opencode: la passphrase passa SOLO via `fd:3` (here-string) — mai come argomento `-pass pass:` (visibile in `ps`) né come file temporaneo. `set -o pipefail` è già attivo: se openssl fallisce (passphrase errata) la pipe ritorna != 0. Le funzioni `_openssl_enc`/`_openssl_dec` sono opzionali; va bene inlinare come sopra.

```bash
is_secret_path() {
  local rel="$1" pat
  for pat in "${SECRET_PATTERNS[@]}"; do
    # shellcheck disable=SC2053
    case "$(basename "$rel")" in $pat) return 0 ;; esac
    case "$rel" in $pat) return 0 ;; esac
  done
  return 1
}

list_repo_untracked() {
  git -C "$1" ls-files --others --exclude-standard 2>/dev/null || true
}

collect_repos() {
  local out="$1"; local first=1
  { printf '['
    local d repo remote branch
    for d in "${REPO_SCAN_DIRS[@]}"; do
      [ -d "$d" ] || continue
      while IFS= read -r -d '' gitdir; do
        repo="$(dirname "$gitdir")"
        remote="$(git -C "$repo" remote get-url origin 2>/dev/null || echo "")"
        branch="$(git -C "$repo" rev-parse --abbrev-ref HEAD 2>/dev/null || echo "")"
        [ $first -eq 1 ] || printf ','
        first=0
        printf '{"path":"%s","remote":"%s","branch":"%s"}' \
          "$repo" "$remote" "$branch"
      done < <(find "$d" -type d -name .git -prune -print0 2>/dev/null)
    done
    printf ']'
  } > "$out"
}
```

```bash
GSUTIL() { "${CONFSYNC_GSUTIL:-gsutil}" "$@"; }

gcs_latest_sha() {
  local ts; ts="$(GSUTIL cat "$BUCKET/$HOSTNAME_TAG/latest" 2>/dev/null || true)"
  [ -n "$ts" ] || { echo ""; return 0; }
  GSUTIL cat "$BUCKET/$HOSTNAME_TAG/$ts/BUNDLE.sha" 2>/dev/null || echo ""
}

gcs_upload() {
  local dir="$1" ts="$2"
  GSUTIL -m cp -r "$dir"/* "$BUCKET/$HOSTNAME_TAG/$ts/" >/dev/null
  printf '%s' "$ts" | GSUTIL cp - "$BUCKET/$HOSTNAME_TAG/latest" >/dev/null
}

cmd_backup() {
  require_deps
  local ts; ts="$(date -u +%Y%m%dT%H%M%SZ)"
  local work; work="${CONFSYNC_WORKDIR_KEEP:-$(mktemp -d)}"; mkdir -p "$work"
  [ -n "${CONFSYNC_WORKDIR_KEEP:-}" ] || trap 'rm -rf "$work"' EXIT

  log "raccolgo dotfiles"
  pack_encrypt "$HOME" "$work/dotfiles.tar.enc" "${INCLUDE_DOTFILES[@]}"

  log "raccolgo secrets"
  # secrets fissi + file sensibili dalle repo (staging in dir dedicata)
  local secstage="$work/.sec"; mkdir -p "$secstage"
  local p
  for p in "${SECRET_PATHS[@]}"; do
    [ -e "$HOME/$p" ] && { mkdir -p "$secstage/$(dirname "$p")"; cp -a "$HOME/$p" "$secstage/$p"; }
  done

  log "scansiono repo"
  collect_repos "$work/repos.manifest.json"
  # file locali repo: non tracciati -> sensibili in secstage, altri in localstage
  local localstage="$work/.local"; mkdir -p "$localstage"
  local repo rel base
  while IFS= read -r repo; do
    [ -n "$repo" ] || continue
    while IFS= read -r rel; do
      [ -n "$rel" ] || continue
      base="${repo#$HOME/}"
      if is_secret_path "$rel"; then
        mkdir -p "$secstage/$base/$(dirname "$rel")"
        cp -a "$repo/$rel" "$secstage/$base/$rel" 2>/dev/null || true
      else
        mkdir -p "$localstage/$base/$(dirname "$rel")"
        cp -a "$repo/$rel" "$localstage/$base/$rel" 2>/dev/null || true
      fi
    done < <(list_repo_untracked "$repo")
  done < <(grep -o '"path":"[^"]*"' "$work/repos.manifest.json" | sed 's/"path":"//;s/"$//')

  pack_encrypt "$secstage"   "$work/secrets.tar.enc"          .
  pack_encrypt "$localstage" "$work/repo-localfiles.tar.enc"  .
  rm -rf "$secstage" "$localstage"

  log "lista pacchetti"
  local plc; plc="$(pkg_list_cmd)"
  if [ -n "$plc" ]; then eval "$plc" > "$work/packages.txt" 2>/dev/null || true; fi
  command -v flatpak >/dev/null 2>&1 && \
    flatpak list --app --columns=application >> "$work/packages.txt" 2>/dev/null || true

  log "manifest + checksum"
  ( cd "$work" && for f in *.tar.enc repos.manifest.json packages.txt; do
      [ -f "$f" ] && printf '%s  %s\n' "$(sha256sum "$f" | awk '{print $1}')" "$f"
    done ) > "$work/MANIFEST.txt"
  sha256sum "$work/MANIFEST.txt" | awk '{print $1}' > "$work/BUNDLE.sha"

  # dedup
  local newsha oldsha; newsha="$(cat "$work/BUNDLE.sha")"; oldsha="$(gcs_latest_sha)"
  if [ "$newsha" = "$oldsha" ] && [ -n "$oldsha" ]; then
    log "nessuna modifica dall'ultimo backup — skip upload (dedup)"
    return 0
  fi
  log "upload -> $BUCKET/$HOSTNAME_TAG/$ts/"
  gcs_upload "$work" "$ts"
  log "backup completato: $ts"
}
```

```bash
make_fs_gsutil() {
  local dir="$1" root="$2"
  cat > "$dir/gsutil" <<EOF
#!/usr/bin/env bash
ROOT="$root"
to_fs() { echo "\$ROOT/\${1#gs://*/}"; }   # strip gs://bucket/
EOF
  cat >> "$dir/gsutil" <<'EOF'
args=("$@"); [ "${args[0]}" = "-m" ] && args=("${args[@]:1}")
cmd="${args[0]}"
case "$cmd" in
  cp)
    # gestisce: cp - dst | cp src dst | cp -r src/* dst/
    if [ "${args[1]}" = "-r" ]; then
      shift 2; dst="$(to_fs "${@: -1}")"; mkdir -p "$dst"
      for s in "${@:1:$#-1}"; do cp -r $s "$dst"/; done
    elif [ "${args[1]}" = "-" ]; then
      dst="$(to_fs "${args[2]}")"; mkdir -p "$(dirname "$dst")"; cat > "$dst"
    else
      src="${args[1]}"; dst="$(to_fs "${args[2]}")"; mkdir -p "$(dirname "$dst")"; cp "$src" "$dst"
    fi ;;
  cat) cat "$(to_fs "${args[1]}")" 2>/dev/null ;;
  ls)  ls "$(to_fs "${args[1]}")" 2>/dev/null ;;
esac
exit 0
EOF
  chmod +x "$dir/gsutil"
}
```

```bash
gcs_fetch_latest() {  # scarica l'ultimo backup in $1
  local dest="$1"; mkdir -p "$dest"
  local ts; ts="$(GSUTIL cat "$BUCKET/$HOSTNAME_TAG/latest" 2>/dev/null || true)"
  [ -n "$ts" ] || die "nessun backup trovato per host $HOSTNAME_TAG"
  GSUTIL -m cp -r "$BUCKET/$HOSTNAME_TAG/$ts/*" "$dest"/ >/dev/null 2>&1 \
    || GSUTIL -m cp -r "$BUCKET/$HOSTNAME_TAG/$ts" "$dest"/ >/dev/null
  echo "$ts"
}

verify_manifest() {
  local dir="$1"
  [ -f "$dir/MANIFEST.txt" ] || die "MANIFEST.txt assente: backup corrotto"
  local sha f
  while read -r sha f; do
    [ -f "$dir/$f" ] || continue
    [ "$(sha256_of "$dir/$f")" = "$sha" ] || die "checksum non valido per $f"
  done < "$dir/MANIFEST.txt"
}

cmd_restore() {
  require_deps
  local do_secrets=0 do_repos=0 do_pkgs=0 dryrun=0 assume_yes=0
  while [ $# -gt 0 ]; do case "$1" in
    --secrets) do_secrets=1 ;; --repos) do_repos=1 ;; --packages) do_pkgs=1 ;;
    --dry-run) dryrun=1 ;; --yes) assume_yes=1 ;; *) die "flag sconosciuta: $1" ;;
  esac; shift; done

  local work; work="$(mktemp -d)"; trap 'rm -rf "$work"' EXIT
  local ts; ts="$(gcs_fetch_latest "$work")"
  log "ripristino backup $ts"
  verify_manifest "$work"

  if [ $dryrun -eq 1 ]; then
    log "[dry-run] estrarrei dotfiles in $HOME; artefatti: $(ls "$work")"; return 0
  fi
  if [ $assume_yes -eq 0 ]; then
    read -rp "Sovrascrivo i dotfiles in $HOME? [y/N] " a; [ "$a" = y ] || die "annullato"
  fi
  [ -f "$work/dotfiles.tar.enc" ] && decrypt_unpack "$work/dotfiles.tar.enc" "$HOME"
  log "dotfiles ripristinati"

  if [ $do_secrets -eq 1 ] && [ -f "$work/secrets.tar.enc" ]; then
    decrypt_unpack "$work/secrets.tar.enc" "$HOME"
    chmod -R go-rwx "$HOME/.ssh" "$HOME/.gnupg" 2>/dev/null || true
    log "secrets ripristinati (permessi 600)"
  fi

  if [ $do_repos -eq 1 ] && [ -f "$work/repos.manifest.json" ]; then
    local entry path remote branch
    grep -o '{[^}]*}' "$work/repos.manifest.json" | while read -r entry; do
      path="$(echo "$entry"  | sed -n 's/.*"path":"\([^"]*\)".*/\1/p')"
      remote="$(echo "$entry"| sed -n 's/.*"remote":"\([^"]*\)".*/\1/p')"
      branch="$(echo "$entry"| sed -n 's/.*"branch":"\([^"]*\)".*/\1/p')"
      [ -n "$remote" ] || { log "skip $path (no remote)"; continue; }
      if [ ! -d "$path/.git" ]; then
        mkdir -p "$(dirname "$path")"
        git clone "$remote" "$path" && [ -n "$branch" ] && git -C "$path" checkout "$branch" 2>/dev/null || true
      fi
    done
    [ -f "$work/repo-localfiles.tar.enc" ] && decrypt_unpack "$work/repo-localfiles.tar.enc" "$HOME"
    log "repo ripristinate"
  fi

  if [ $do_pkgs -eq 1 ] && [ -f "$work/packages.txt" ]; then
    local distro; distro="$(detect_distro)"
    local inst; case "$distro" in
      arch) inst="sudo pacman -S --needed -" ;;
      debian) inst="sudo xargs apt-get install -y <" ;;
      fedora) inst="sudo xargs dnf install -y <" ;;
      *) inst="" ;;
    esac
    log "pacchetti da reinstallare (distro: $distro). Comando suggerito:"
    printf '  %s packages.txt\n' "${inst:-<manuale>}" >&2
    log "(esecuzione manuale per sicurezza)"
  fi
  log "restore completato"
}

cmd_list() {
  require_deps
  GSUTIL ls "$BUCKET/$HOSTNAME_TAG/" 2>/dev/null || die "nessun backup"
}

cmd_diff() {
  require_deps
  local work; work="$(mktemp -d)"; trap 'rm -rf "$work"' EXIT
  gcs_fetch_latest "$work" >/dev/null
  local ext="$work/ext"; mkdir -p "$ext"
  [ -f "$work/dotfiles.tar.enc" ] && decrypt_unpack "$work/dotfiles.tar.enc" "$ext"
  log "differenze dotfiles (backup vs locale):"
  ( cd "$ext" && find . -type f | while read -r f; do
      diff -q "$f" "$HOME/${f#./}" >/dev/null 2>&1 || echo "DIVERSO: ${f#./}"
    done ) >&2 || true
}
```
